cmake_minimum_required(VERSION 3.21)
project(Vision3DLocalAI VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Vision3DLocalAI currently targets Windows x64")
endif()

find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
add_executable(Vision3DLocalAI src/main.cpp src/JobProcessor.cpp src/JobProcessor.h)
target_compile_features(Vision3DLocalAI PRIVATE cxx_std_17)
target_link_libraries(Vision3DLocalAI PRIVATE Qt6::Core)
target_compile_definitions(Vision3DLocalAI PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(Vision3DLocalAI PRIVATE /W4 /utf-8)
endif()
